📄 MCP_Server.gs (Model Context Protocol Bridge)

              
            

            
/**
 * RAZUM SOVEREIGN AI - GAS MCP SERVER v2.6
 * Production Model Context Protocol Server for Google Workspace
 */

function doPost(e) {
  try {
    const request = JSON.parse(e.postData.contents);
    const method = request.method;
    const params = request.params || {};

    let responseData = {};

    switch (method) {
      case "tools/list":
        responseData = {
          tools: [
            { name: "read_sheet", description: "Чтение реестра из Google Таблицы" },
            { name: "append_invoice", description: "Запись разобранного счета в реестр" },
            { name: "create_draft_email", description: "Создание черновика письма в Gmail" },
            { name: "schedule_calendar", description: "Бронирование слота в Google Calendar" }
          ]
        };
        break;

      case "tools/call":
        responseData = handleToolCall(params.name, params.arguments);
        break;

      default:
        throw new Error("Unknown MCP method: " + method);
    }

    return ContentService.createTextOutput(JSON.stringify({ result: responseData }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
